public class MainClass {
  
  static Integer prime_n (int N){
    
    int x,y,flg;
    
    for(x = 1; x <= N; x++){
      
      if(x == 1 || x == 0){
        continue;
        flg = 1;
        }
    }
        
    for (y = 2; y <= x/2; ++y){
      if(x%y == 0){
        flg = 0;
        break;
        }
    }
    
  }
      
  public static void main(String[] args) {
    
    int N = 15;
    
    int Prime = prime_n(N);
    
    System.out.println("The prime numbers from 1 to " + N + " is " + Prime);
    
    
  }
}
