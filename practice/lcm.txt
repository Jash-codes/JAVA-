public class MainClass {
  
  static int findlcm (int a, int b){
    
    int lcm = (a > b) ? a : b;
    while(true){
      if (lcm % a == 0 && lcm%b == 0){
        break;  
    }
    lcm++;
  }
    return lcm;
  }
      
  public static void main(String[] args) {
    
    int a = 12;
    int b = 16;
    
    int LCM = findlcm(a,b);
    
    System.out.println("The LCM of " + a + " and " + b + " is " + LCM);
  }
}
