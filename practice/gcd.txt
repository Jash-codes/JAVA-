public class MainClass {
  
  static int findgcd (int x, int y){
    
    if(y == 0){
      return x;
    }
    
    return findgcd(y,x%y);
    
  }
      
  public static void main(String[] args) {
    
    int x = 12;
    int y = 16;
    
    int GCD = findgcd(x,y);
    
    System.out.println("The GCD of " + x + " and " + y + " is " + GCD); 
    
  }
}
